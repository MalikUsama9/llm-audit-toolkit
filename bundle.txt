
===== src\audit\llm\client.py =====

"""Provider-agnostic LLM client.

Every LLM call in the project goes through LLMClient.complete().
It handles: timeout, retries, caching, cost/token/latency logging, budget cap.
Swapping the model is just a different `model` string (e.g. "gemini/gemini-2.5-flash").
"""
from __future__ import annotations

import asyncio
import hashlib
import json
import sqlite3
import threading
import time
from dataclasses import asdict, dataclass
from pathlib import Path

import litellm
from dotenv import load_dotenv

load_dotenv()  # reads GEMINI_API_KEY / OPENAI_API_KEY from your .env file

# Errors where retrying cannot help (bad key, bad request, unknown model)
NO_RETRY = tuple(
    getattr(litellm, name)
    for name in ("AuthenticationError", "BadRequestError", "NotFoundError")
    if hasattr(litellm, name)
)


class BudgetExceeded(RuntimeError):
    """Raised when total spend passes the budget cap."""


@dataclass
class LLMResponse:
    text: str
    model: str
    input_tokens: int
    output_tokens: int
    cost_usd: float
    latency_ms: float
    cached: bool = False


class LLMClient:
    def __init__(self, db_path: str = "data/llm_cache.db", budget_usd: float = 5.0,
                 timeout_s: float = 180.0):
        Path(db_path).parent.mkdir(parents=True, exist_ok=True)
        self.db = sqlite3.connect(db_path, check_same_thread=False)
        self.db.execute("CREATE TABLE IF NOT EXISTS cache (k TEXT PRIMARY KEY, v TEXT)")
        self.db.execute("""CREATE TABLE IF NOT EXISTS call_log (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            ts TEXT DEFAULT CURRENT_TIMESTAMP,
            model TEXT, input_tokens INTEGER, output_tokens INTEGER,
            cost_usd REAL, latency_ms REAL, cached INTEGER, error TEXT)""")
        self.db.commit()
        self.lock = threading.Lock()
        self.budget = budget_usd
        self.timeout_s = timeout_s
        self.spent = 0.0  # spend in this session (cache hits cost nothing)

    # ---------- public API ----------
    def complete(self, messages: list[dict], model: str, temperature: float = 0.0,
                 max_tokens: int = 700, retries: int = 4,
                 use_cache: bool = True) -> LLMResponse:
        key = self._key(messages, model, temperature, max_tokens)

        if use_cache:
            hit = self._cache_get(key)
            if hit:
                hit.cached = True
                self._log(hit.model, hit.input_tokens, hit.output_tokens, 0.0,
                          hit.latency_ms, cached=True)
                return LLMResponse(**{**asdict(hit), "cost_usd": 0.0})

        if self.spent >= self.budget:
            raise BudgetExceeded(f"spent ${self.spent:.4f} of ${self.budget:.2f}")

        resp, latency_ms = self._call_with_retries(
            messages, model, temperature, max_tokens, retries)

        try:
            cost = litellm.completion_cost(completion_response=resp) or 0.0
        except Exception:
            cost = 0.0  # unknown price: log 0, don't crash

        usage = resp.usage
        out = LLMResponse(
            text=resp.choices[0].message.content or "",
            model=model,
            input_tokens=getattr(usage, "prompt_tokens", 0) or 0,
            output_tokens=getattr(usage, "completion_tokens", 0) or 0,
            cost_usd=cost,
            latency_ms=latency_ms,
        )
        with self.lock:
            self.spent += cost
        self._cache_put(key, out)
        self._log(model, out.input_tokens, out.output_tokens, cost, latency_ms, cached=False)
        return out

    async def acomplete(self, *args, **kwargs) -> LLMResponse:
        """Async wrapper, so the runner can make many calls concurrently."""
        return await asyncio.to_thread(self.complete, *args, **kwargs)

    def total_logged_cost(self) -> float:
        with self.lock:
            row = self.db.execute("SELECT COALESCE(SUM(cost_usd), 0) FROM call_log").fetchone()
        return row[0]

    # ---------- internals ----------
    def _call_with_retries(self, messages, model, temperature, max_tokens, retries):
        last_err = None
        for attempt in range(retries):
            start = time.perf_counter()
            try:
                resp = litellm.completion(
                    model=model, messages=messages, temperature=temperature,
                    max_tokens=max_tokens, timeout=self.timeout_s)
                return resp, (time.perf_counter() - start) * 1000
            except NO_RETRY as e:
                self._log(model, 0, 0, 0.0, 0.0, cached=False, error=repr(e))
                raise
            except Exception as e:
                last_err = e
                self._log(model, 0, 0, 0.0, (time.perf_counter() - start) * 1000,
                          cached=False, error=repr(e))
                if attempt < retries - 1:
                    time.sleep(min(5 * 3 ** attempt, 60))  # 5s, 15s, 45s...
        raise last_err

    @staticmethod
    def _key(messages, model, temperature, max_tokens) -> str:
        raw = json.dumps([messages, model, temperature, max_tokens], sort_keys=True)
        return hashlib.sha256(raw.encode()).hexdigest()

    def _cache_get(self, key) -> LLMResponse | None:
        with self.lock:
            row = self.db.execute("SELECT v FROM cache WHERE k=?", (key,)).fetchone()
        return LLMResponse(**json.loads(row[0])) if row else None

    def _cache_put(self, key, out: LLMResponse) -> None:
        with self.lock:
            self.db.execute("INSERT OR REPLACE INTO cache VALUES (?, ?)",
                            (key, json.dumps(asdict(out))))
            self.db.commit()

    def _log(self, model, in_tok, out_tok, cost, latency_ms, cached, error=None) -> None:
        with self.lock:
            self.db.execute(
                "INSERT INTO call_log (model, input_tokens, output_tokens, cost_usd,"
                " latency_ms, cached, error) VALUES (?,?,?,?,?,?,?)",
                (model, in_tok, out_tok, cost, latency_ms, int(cached), error))
            self.db.commit()

===== src\audit\checks.py =====

"""Deterministic checks that run BEFORE the LLM judge."""
import re
from dataclasses import dataclass

REFUSAL_PATTERNS = [
    # "I can't / cannot / am unable to ... answer/help/provide/assist"
    r"\bi (?:can(?:no|')t|can not|cannot|am unable to)\b.{0,40}\b(?:answer|help|provide|assist|find)",
    # "I'm unable to" / "I'm not able to"
    r"\bi(?:'m| am) (?:not able|unable) to\b",
    # "I don't know" / "I don't have that information / the details / enough context"
    r"\bi (?:don't|do not) know\b",
    r"\bi (?:don't|do not) have\b.{0,30}?\b(?:information|info|details|data|context)\b",
    # "not mentioned in the context"
    r"\bnot (?:mentioned|specified|found|available) in (?:the |my )?(?:provided )?(?:context|documents?)",
    r"\bnot (?:in|within) my (?:context|knowledge)\b",
    r"\bno (?:relevant )?information (?:is )?(?:available|provided|found)",
    r"\bi'?m sorry,? but\b",
    r"\bunable to (?:find|answer)\b",
]

ERROR_PATTERNS = [
    r"^\s*(error|exception|traceback)\b",
    r"\b(rate limit|timed? ?out|connection (error|refused)|internal server error)\b",
    r"\b(api key|authentication) (error|failed|invalid)\b",
]

CITATION_PATTERNS = [
    r"\[\d+\]",                # [1]
    r"\(source:[^)]+\)",       # (source: doc.pdf)
    r"\bsource[s]?:",          # Source: ...
]

# Statuses that mean "do not send to the judge as a normal answer"
SKIP_JUDGE_STATUSES = {"empty", "error"}


@dataclass
class CheckResult:
    status: str          # "ok" | "empty" | "error" | "refusal" | "partial"
    has_citation: bool
    detail: str = ""


def _normalize(text: str) -> str:
    """Turn curly quotes into straight ones so patterns like "don't" always match."""
    return text.replace("\u2019", "'").replace("\u2018", "'")


def is_empty(answer) -> bool:
    return answer is None or not str(answer).strip()


def is_error(answer: str) -> bool:
    return any(re.search(p, answer, re.I) for p in ERROR_PATTERNS)


def is_refusal(answer: str) -> bool:
    return any(re.search(p, answer, re.I | re.S) for p in REFUSAL_PATTERNS)


def has_citation(answer: str) -> bool:
    return any(re.search(p, answer, re.I) for p in CITATION_PATTERNS)


def run_checks(answer, bot_error=None) -> CheckResult:
    """Order matters: error/empty first, then refusal/partial."""
    if bot_error:
        return CheckResult("error", False, f"bot raised: {bot_error}")
    if is_empty(answer):
        return CheckResult("empty", False, "empty answer")
    answer = _normalize(str(answer))
    if is_error(answer) and len(answer) < 300:   # short + error words = likely an error message
        return CheckResult("error", False, "looks like an error message")
    cite = has_citation(answer)
    if is_refusal(answer):
        if cite:
            # Real, cited answer that declines only part of the question
            return CheckResult("partial", cite, "cited answer, then declined part")
        return CheckResult("refusal", cite, "matched refusal pattern")
    return CheckResult("ok", cite)

===== src\audit\store.py =====

"""SQLite store: runs, cases, results, checks, scores.

One database file holds everything for an audit, so runs can be compared later.
Rows go in and out as the Pydantic models from models.py.
"""
from __future__ import annotations

import json
import sqlite3
import threading
from datetime import datetime
from pathlib import Path

from audit.models import Result, Run, Score, TestCase

SCHEMA = """
CREATE TABLE IF NOT EXISTS runs (
    run_id          TEXT PRIMARY KEY,
    target_name     TEXT NOT NULL,
    config_snapshot TEXT NOT NULL DEFAULT '{}',   -- JSON
    created_at      TEXT NOT NULL,                -- ISO timestamp
    notes           TEXT
);

CREATE TABLE IF NOT EXISTS cases (
    id                TEXT PRIMARY KEY,
    category          TEXT NOT NULL,
    question          TEXT NOT NULL,
    reference_answer  TEXT,
    source_ref        TEXT,                       -- JSON list, or NULL
    expected_behavior TEXT NOT NULL,
    difficulty        TEXT NOT NULL,
    source_text       TEXT,
    origin            TEXT NOT NULL DEFAULT 'generated',
    reviewed          INTEGER NOT NULL DEFAULT 0,
    split             TEXT
);

CREATE TABLE IF NOT EXISTS results (
    run_id        TEXT NOT NULL REFERENCES runs(run_id),
    case_id       TEXT NOT NULL REFERENCES cases(id),
    answer        TEXT,
    contexts      TEXT,                           -- JSON list, or NULL
    latency_ms    REAL,
    input_tokens  INTEGER,
    output_tokens INTEGER,
    error         TEXT,
    PRIMARY KEY (run_id, case_id)
);

CREATE TABLE IF NOT EXISTS scores (
    run_id      TEXT NOT NULL,
    case_id     TEXT NOT NULL,
    metric      TEXT NOT NULL,
    value       REAL NOT NULL,
    reasoning   TEXT,
    judge_model TEXT,
    PRIMARY KEY (run_id, case_id, metric),
    FOREIGN KEY (run_id, case_id) REFERENCES results(run_id, case_id)
);

-- Deterministic check result for each answer. Kept apart from `scores` so that
-- empty / error / refusal rows are reported as their own row and never averaged
-- in as a score of 0.
CREATE TABLE IF NOT EXISTS checks (
    run_id       TEXT NOT NULL,
    case_id      TEXT NOT NULL,
    status       TEXT NOT NULL,                  -- ok | empty | error | refusal | partial
    has_citation INTEGER NOT NULL DEFAULT 0,
    detail       TEXT,
    PRIMARY KEY (run_id, case_id),
    FOREIGN KEY (run_id, case_id) REFERENCES results(run_id, case_id)
);

CREATE INDEX IF NOT EXISTS idx_results_run ON results(run_id);
CREATE INDEX IF NOT EXISTS idx_scores_run  ON scores(run_id);
CREATE INDEX IF NOT EXISTS idx_checks_run  ON checks(run_id);
"""

# Columns added to `cases` after the first version of the schema.
# Used to upgrade a database file that was created before they existed.
_NEW_CASE_COLUMNS = {
    "source_text": "TEXT",
    "origin": "TEXT NOT NULL DEFAULT 'generated'",
    "reviewed": "INTEGER NOT NULL DEFAULT 0",
    "split": "TEXT",
}


class Store:
    def __init__(self, db_path: str = "data/audit.db"):
        if db_path != ":memory:":
            Path(db_path).parent.mkdir(parents=True, exist_ok=True)
        self.db = sqlite3.connect(db_path, check_same_thread=False)
        self.db.row_factory = sqlite3.Row
        self.db.execute("PRAGMA foreign_keys = ON")
        self.db.executescript(SCHEMA)
        self._migrate()
        self.lock = threading.Lock()

    def _migrate(self) -> None:
        """Add any missing columns to an older `cases` table."""
        existing = {row["name"] for row in self.db.execute("PRAGMA table_info(cases)")}
        for name, definition in _NEW_CASE_COLUMNS.items():
            if name not in existing:
                self.db.execute(f"ALTER TABLE cases ADD COLUMN {name} {definition}")
        self.db.commit()

    def close(self) -> None:
        self.db.close()

    # ---------- runs ----------
    def add_run(self, run: Run) -> None:
        self._write(
            "INSERT INTO runs VALUES (?,?,?,?,?)",
            (run.run_id, run.target_name, json.dumps(run.config_snapshot),
             run.created_at.isoformat(), run.notes))

    def get_run(self, run_id: str) -> Run | None:
        row = self._one("SELECT * FROM runs WHERE run_id=?", (run_id,))
        return self._row_to_run(row) if row else None

    def list_runs(self) -> list[Run]:
        return [self._row_to_run(r) for r in
                self._all("SELECT * FROM runs ORDER BY created_at")]

    # ---------- cases ----------
    def add_cases(self, cases: list[TestCase]) -> None:
        """Insert cases; an existing case with the same id is replaced."""
        rows = [
            (c.id, c.category, c.question, c.reference_answer,
             json.dumps(c.source_ref) if c.source_ref is not None else None,
             c.expected_behavior.value, c.difficulty.value,
             c.source_text, c.origin.value, int(c.reviewed),
             c.split.value if c.split is not None else None)
            for c in cases
        ]
        with self.lock:
            self.db.executemany(
                "INSERT OR REPLACE INTO cases "
                "(id, category, question, reference_answer, source_ref, "
                "expected_behavior, difficulty, source_text, origin, reviewed, split) "
                "VALUES (?,?,?,?,?,?,?,?,?,?,?)",
                rows)
            self.db.commit()

    def get_cases(self, category: str | None = None) -> list[TestCase]:
        if category:
            rows = self._all("SELECT * FROM cases WHERE category=? ORDER BY id", (category,))
        else:
            rows = self._all("SELECT * FROM cases ORDER BY id")
        out = []
        for row in rows:
            d = dict(row)
            d["source_ref"] = self._parse_source_ref(d["source_ref"])
            d["reviewed"] = bool(d["reviewed"])
            out.append(TestCase(**d))
        return out

    # ---------- results ----------
    def add_result(self, r: Result) -> None:
        """Insert a result; re-running the same case in the same run replaces it."""
        self._write(
            "INSERT OR REPLACE INTO results VALUES (?,?,?,?,?,?,?,?)",
            (r.run_id, r.case_id, r.answer,
             json.dumps(r.contexts) if r.contexts is not None else None,
             r.latency_ms, r.input_tokens, r.output_tokens, r.error))

    def get_results(self, run_id: str) -> list[Result]:
        rows = self._all("SELECT * FROM results WHERE run_id=? ORDER BY case_id", (run_id,))
        out = []
        for row in rows:
            d = dict(row)
            d["contexts"] = json.loads(d["contexts"]) if d["contexts"] else None
            out.append(Result(**d))
        return out

    def done_case_ids(self, run_id: str) -> set[str]:
        """Cases already answered without error. Lets the runner resume a run."""
        rows = self._all(
            "SELECT case_id FROM results WHERE run_id=? AND error IS NULL", (run_id,))
        return {r["case_id"] for r in rows}

    # ---------- checks ----------
    def add_check(self, run_id: str, case_id: str, status: str,
                  has_citation: bool, detail: str = "") -> None:
        """Save the deterministic check result for one answer (replaces an old one)."""
        self._write(
            "INSERT OR REPLACE INTO checks VALUES (?,?,?,?,?)",
            (run_id, case_id, status, int(has_citation), detail))

    def get_checks(self, run_id: str) -> dict[str, dict]:
        """case_id -> check row. The judge uses this to skip empty/error answers."""
        rows = self._all("SELECT * FROM checks WHERE run_id=?", (run_id,))
        return {r["case_id"]: dict(r) for r in rows}

    # ---------- scores ----------
    def add_score(self, s: Score) -> None:
        """Insert a score; re-scoring the same metric replaces the old one."""
        self._write(
            "INSERT OR REPLACE INTO scores VALUES (?,?,?,?,?,?)",
            (s.run_id, s.case_id, s.metric, s.value, s.reasoning, s.judge_model))

    def get_scores(self, run_id: str, metric: str | None = None) -> list[Score]:
        if metric:
            rows = self._all(
                "SELECT * FROM scores WHERE run_id=? AND metric=? ORDER BY case_id",
                (run_id, metric))
        else:
            rows = self._all("SELECT * FROM scores WHERE run_id=? ORDER BY case_id, metric",
                             (run_id,))
        return [Score(**dict(r)) for r in rows]

    # ---------- helpers ----------
    def _write(self, sql: str, params: tuple) -> None:
        with self.lock:
            self.db.execute(sql, params)
            self.db.commit()

    def _one(self, sql: str, params: tuple = ()):
        with self.lock:
            return self.db.execute(sql, params).fetchone()

    def _all(self, sql: str, params: tuple = ()):
        with self.lock:
            return self.db.execute(sql, params).fetchall()

    @staticmethod
    def _parse_source_ref(raw: str | None) -> list[str] | None:
        """Read source_ref back as a list.

        New rows hold a JSON list. Rows saved before this change hold a plain
        string like 'policy.md#1', which becomes a one-item list.
        """
        if raw is None:
            return None
        try:
            value = json.loads(raw)
        except json.JSONDecodeError:
            return [raw]
        if isinstance(value, list):
            return value
        return [raw]

    @staticmethod
    def _row_to_run(row) -> Run:
        return Run(run_id=row["run_id"], target_name=row["target_name"],
                   config_snapshot=json.loads(row["config_snapshot"]),
                   created_at=datetime.fromisoformat(row["created_at"]),
                   notes=row["notes"])

===== src\audit\runner.py =====

"""Runner: asks the target every test question and saves each result as it finishes."""
from __future__ import annotations

import asyncio
import time
import uuid
from collections import Counter
from datetime import datetime

from audit.models import Result, Run
from audit.store import Store
from audit.targets.base import Target, TargetResponse

# Failure categories. They go in Result.error, so a scorer can skip them
# instead of counting them as wrong answers.
EMPTY = "empty_answer"
TIMEOUT = "timeout"
ERROR = "error"


async def _ask_once(target: Target, question: str, timeout_s: float) -> TargetResponse:
    start = time.perf_counter()
    # target.answer is a normal (blocking) function, so run it in a worker thread
    resp = await asyncio.wait_for(asyncio.to_thread(target.answer, question),
                                  timeout=timeout_s)
    if resp.latency_ms is None:
        resp.latency_ms = (time.perf_counter() - start) * 1000
    return resp


async def _ask_with_retries(target: Target, question: str, timeout_s: float,
                            max_retries: int, base_delay: float) -> TargetResponse:
    last_error = None
    for attempt in range(max_retries + 1):
        try:
            resp = await _ask_once(target, question, timeout_s)
            if resp.error is None:
                return resp
            last_error = f"{ERROR}: {resp.error}"
        except asyncio.TimeoutError:
            last_error = f"{TIMEOUT}: no answer within {timeout_s}s"
        except Exception as e:                       # one bad case must not kill the run
            last_error = f"{ERROR}: {type(e).__name__}: {e}"
        if attempt < max_retries:
            await asyncio.sleep(base_delay * 2 ** attempt)   # 1s, 2s, 4s, ...
    return TargetResponse(error=last_error)


async def _run_case(sem: asyncio.Semaphore, store: Store, target: Target,
                    run_id: str, case, timeout_s: float,
                    max_retries: int, base_delay: float) -> None:
    async with sem:                                  # at most N cases in flight
        # Leak rule: the target gets ONLY the question.
        resp = await _ask_with_retries(target, case.question, timeout_s,
                                       max_retries, base_delay)

    error = resp.error
    if error is None and not (resp.answer or "").strip():
        error = EMPTY                                # empty is its own category

    store.add_result(Result(                         # saved immediately
        run_id=run_id, case_id=case.id, answer=resp.answer,
        contexts=resp.contexts, latency_ms=resp.latency_ms,
        input_tokens=resp.input_tokens, output_tokens=resp.output_tokens,
        error=error))


async def run_audit(store: Store, target: Target, cases: list, *,
                    run_id: str | None = None, config_snapshot: dict | None = None,
                    concurrency: int = 4, timeout_s: float = 60.0,
                    max_retries: int = 3, base_delay: float = 1.0,
                    notes: str | None = None) -> dict:
    """Run every case through the target. Pass an old run_id to resume that run."""
    run_id = run_id or f"run_{datetime.now():%Y%m%d_%H%M%S}_{uuid.uuid4().hex[:6]}"

    if store.get_run(run_id) is None:
        store.add_run(Run(run_id=run_id, target_name=target.name,
                          config_snapshot=config_snapshot or {},
                          created_at=datetime.now(), notes=notes))

    done = store.done_case_ids(run_id)               # resume: skip finished cases
    todo = [c for c in cases if c.id not in done]

    sem = asyncio.Semaphore(concurrency)
    outcomes = await asyncio.gather(
        *(_run_case(sem, store, target, run_id, c, timeout_s, max_retries, base_delay)
          for c in todo),
        return_exceptions=True)                      # safety net: never crash the run

    results = store.get_results(run_id)
    kinds = Counter("ok" if r.error is None else r.error.split(":")[0] for r in results)
    return {"run_id": run_id, "skipped": len(done), "attempted": len(todo),
            "internal_errors": sum(isinstance(o, Exception) for o in outcomes),
            "totals": dict(kinds)}

===== src\audit\models.py =====

"""Core data models for the audit toolkit.

Everything else (runner, evaluators, store, reports) depends on these,
so keep field names stable. Add new optional fields rather than renaming.
"""
from __future__ import annotations

from datetime import datetime, timezone
from enum import Enum
from typing import Any

from pydantic import BaseModel, Field, field_validator, model_validator


class ExpectedBehavior(str, Enum):
    ANSWER = "answer"    # bot should answer from the docs
    REFUSE = "refuse"    # out of scope / unsafe: bot should decline
    HANDOFF = "handoff"  # bot should pass to a human
    CLARIFY = "clarify"  # question is ambiguous: bot should ask a follow-up


class Difficulty(str, Enum):
    EASY = "easy"
    MEDIUM = "medium"
    HARD = "hard"


class Origin(str, Enum):
    GENERATED = "generated"  # written by an LLM
    HUMAN = "human"          # written or rewritten by you


class Split(str, Enum):
    DEV = "dev"
    HELDOUT = "heldout"


class TestCase(BaseModel):
    __test__ = False  # stops pytest from trying to collect this class

    id: str
    category: str                        # e.g. "factual", "off_topic", "adversarial"
    question: str
    reference_answer: str | None = None  # None for refuse/handoff/clarify cases
    source_ref: list[str] | None = None  # doc names + chunk/page the answer comes from
    source_text: str | None = None       # exact passage the answer is based on
    expected_behavior: ExpectedBehavior = ExpectedBehavior.ANSWER
    difficulty: Difficulty = Difficulty.MEDIUM
    origin: Origin = Origin.GENERATED
    reviewed: bool = False               # has a human checked this case?
    split: Split | None = None           # dev / heldout

    @field_validator("question")
    @classmethod
    def question_not_empty(cls, v: str) -> str:
        if not v.strip():
            raise ValueError("question must not be empty")
        return v.strip()

    @field_validator("source_ref", mode="before")
    @classmethod
    def source_ref_to_list(cls, v):
        # Old data stored a single string; wrap it so it still loads
        if isinstance(v, str):
            return [v]
        return v

    @model_validator(mode="after")
    def answerable_needs_source(self) -> "TestCase":
        # Roadmap rule: every answerable case has a source reference
        if self.expected_behavior == ExpectedBehavior.ANSWER and not self.source_ref:
            raise ValueError(f"case {self.id}: answerable cases need a source_ref")
        return self


class Result(BaseModel):
    run_id: str
    case_id: str
    answer: str | None = None
    contexts: list[str] | None = None   # only in glass-box mode
    latency_ms: float | None = None
    input_tokens: int | None = None

"""Load and validate the audit settings (a YAML file).

Typos in the YAML are rejected on purpose (extra="forbid"), so a misspelled key
fails loudly instead of being silently ignored.
"""
from __future__ import annotations

from pathlib import Path
from typing import Literal

import yaml
from pydantic import BaseModel, ConfigDict, Field, field_validator


class _Strict(BaseModel):
    model_config = ConfigDict(extra="forbid")


class TargetConfig(_Strict):
    name: str
    type: Literal["python", "http", "file"] = "python"
    entrypoint: str | None = None   # python: path to the bot file
    model: str | None = None        # model the bot uses (recorded for the report)
    # http target
    url: str | None = None
    headers: dict[str, str] | None = None
    payload_template: dict | None = None
    answer_path: str | None = None
    contexts_path: str | None = None
    timeout_s: float | None = None
    # file target
    answers_file: str | None = None  # CSV the client filled in


class ModelsConfig(_Strict):
    generator: str                  # writes the test questions
    judge: str                      # scores the bot's answers

    @field_validator("generator", "judge")
    @classmethod
    def not_empty(cls, v: str) -> str:
        if not v.strip():
            raise ValueError("model name must not be empty")
        return v.strip()


class JudgeConfig(_Strict):
    """Settings for the judge that scores the bot's answers."""
    model: str                                   # e.g. groq/openai/gpt-oss-120b
    temperature: float = Field(default=0.0, ge=0.0, le=2.0)
    max_tokens: int = Field(default=800, ge=1)
    rubric_dir: str = "configs/rubrics"
    budget_usd: float = Field(default=3.0, gt=0)  # spending cap for judging

    @field_validator("model")
    @classmethod
    def model_not_empty(cls, v: str) -> str:
        if not v.strip():
            raise ValueError("judge model name must not be empty")
        return v.strip()


class BudgetConfig(_Strict):
    max_usd: float = Field(gt=0, description="hard spending cap for one audit")
    timeout_s: float = Field(default=180.0, gt=0)
    max_concurrency: int = Field(default=4, ge=1, le=64)


class PathsConfig(_Strict):
    docs_dir: str = "data/demo/docs"
    testset: str = "data/demo/testset.jsonl"
    database: str = "data/audit.db"
    llm_cache: str = "data/llm_cache.db"


class TestsetConfig(_Strict):
    __test__ = False  # not a pytest class
    holdout_fraction: float = Field(default=0.3, ge=0.0, le=0.5)
    categories: dict[str, int]

    @field_validator("categories")
    @classmethod
    def counts_positive(cls, v: dict[str, int]) -> dict[str, int]:
        if not v:
            raise ValueError("at least one category is required")
        bad = [k for k, n in v.items() if n < 1]
        if bad:
            raise ValueError(f"category counts must be at least 1: {bad}")
        return v

    @property
    def total(self) -> int:
        return sum(self.categories.values())


class AuditConfig(_Strict):
    target: TargetConfig
    models: ModelsConfig
    judge: JudgeConfig | None = None
    budget: BudgetConfig
    paths: PathsConfig = PathsConfig()
    testset: TestsetConfig

    def warnings(self) -> list[str]:
        """Things that are allowed but weaken the audit. Show them in the report."""
        out = []
        provider = lambda m: m.split("/", 1)[0] if m else None  # noqa: E731
        if self.target.model and provider(self.models.judge) == provider(self.target.model):
            out.append("Judge and target bot use the same provider; "
                       "scores may be biased toward the bot's own style.")
        if self.models.generator == self.models.judge:
            out.append("Test generator and judge are the same model; note this in the limitations.")
        if self.judge and self.judge.model != self.models.judge:
            out.append(f"judge.model ({self.judge.model}) differs from models.judge "
                       f"({self.models.judge}); keep them identical.")
        return out


def load_config(path: str | Path) -> AuditConfig:
    p = Path(path)
    if not p.exists():
        raise FileNotFoundError(f"Config file not found: {p}")
    try:
        data = yaml.safe_load(p.read_text(encoding="utf-8"))
    except yaml.YAMLError as e:
        raise ValueError(f"{p} is not valid YAML: {e}") from e
    if not isinstance(data, dict):
        raise ValueError(f"{p} is empty or not a YAML mapping")
    return AuditConfig.model_validate(data)

===== src\audit\cli.py =====

"""Command line: `audit run` and `audit check`.

    audit run --target configs/audit.yaml --testset data/demo/testset.jsonl --limit 5
    audit run --resume run_20261105_120000_ab12cd     # continue an earlier run
    audit check --run run_20261105_120000_ab12cd      # deterministic checks

`run` reads the config, builds the target, loads the test cases, calls run_audit(),
prints progress while it works, and a summary (errors, p50/p95 latency) at the end.

`check` runs the deterministic checks (empty, error, refusal, citation) on the saved
answers of a run and stores each status in the `checks` table. These run before the
judge. Empty and error answers are reported as their own row, never scored as wrong.
"""
from __future__ import annotations

import argparse
import asyncio
import json
import sys
import uuid
from collections import Counter
from datetime import datetime
from pathlib import Path

from pydantic import ValidationError

from audit.checks import run_checks
from audit.config import AuditConfig, load_config
from audit.models import TestCase
from audit.runner import run_audit
from audit.store import Store
from audit.targets.base import Target


# ---------------------------------------------------------------- helpers
def percentile(values: list[float], p: float) -> float:
    """p in 0..100. Linear interpolation between the closest ranks."""
    if not values:
        return 0.0
    s = sorted(values)
    k = (len(s) - 1) * p / 100
    lo, hi = int(k), min(int(k) + 1, len(s) - 1)
    return s[lo] + (s[hi] - s[lo]) * (k - lo)


def load_testset(path: str | Path) -> list[TestCase]:
    """Read a .jsonl file: one test case per line."""
    p = Path(path)
    if not p.exists():
        raise FileNotFoundError(f"Test set not found: {p}")
    cases = []
    with p.open(encoding="utf-8") as f:
        for n, line in enumerate(f, start=1):
            line = line.strip()
            if not line:
                continue
            try:
                cases.append(TestCase(**json.loads(line)))
            except (json.JSONDecodeError, ValidationError) as e:
                raise ValueError(f"{p}, line {n}: {e}") from e
    return cases


def build_target(cfg: AuditConfig) -> Target:
    """Pick PythonTarget / HttpTarget / FileTarget from `target.type` in the YAML."""
    t = cfg.target
    if t.type == "python":
        from audit.llm.client import LLMClient
        from audit.targets.python_target import PythonTarget
        # The spending cap and cache path come from the YAML, not the client defaults.
        client = LLMClient(
            db_path=cfg.paths.llm_cache,
            budget_usd=cfg.budget.max_usd,
        )
        target = PythonTarget.from_rag_bot(client, name=t.name)
        if t.model:                      # the model named in the YAML wins
            target.config.model = t.model
        return target
    if t.type == "http":
        from audit.targets.http_target import HttpTarget
        if not t.url:
            raise ValueError("target.type is 'http' but target.url is missing in the YAML")
        return HttpTarget.from_config(t.model_dump(exclude_none=True))
    if t.type == "file":
        from audit.targets.file_target import FileTarget
        if not t.answers_file:
            raise ValueError("target.type is 'file' but target.answers_file is missing")
        return FileTarget.from_csv(t.answers_file, name=t.name)
    raise ValueError(f"unknown target type: {t.type}")


async def _show_progress(store: Store, run_id: str, total: int, every_s: float = 2.0) -> None:
    """Print 'done/total (errors)' every few seconds while the run is going."""
    last = -1
    while True:
        await asyncio.sleep(every_s)
        results = store.get_results(run_id)
        done = len(results)
        if done != last:
            errors = sum(1 for r in results if r.error is not None)
            print(f"  progress: {done}/{total} saved, {errors} errors", flush=True)
            last = done


def print_summary(store: Store, run_id: str, case_ids: set[str], outcome: dict) -> int:
    """Print the end-of-run summary. Returns the number of errors."""
    results = [r for r in store.get_results(run_id) if r.case_id in case_ids]
    errors = [r for r in results if r.error is not None]
    kinds: dict[str, int] = {}
    for r in errors:
        kind = r.error.split(":")[0]
        kinds[kind] = kinds.get(kind, 0) + 1
    latencies = [r.latency_ms for r in results if r.error is None and r.latency_ms is not None]

    print()
    print("=" * 50)
    print(f"Run:        {run_id}")
    print(f"Cases:      {len(results)} saved (skipped {outcome['skipped']} already done, "
          f"asked {outcome['attempted']} this time)")
    print(f"Succeeded:  {len(results) - len(errors)}")
    print(f"Errors:     {len(errors)}" + (f"  {kinds}" if kinds else ""))
    if latencies:
        print(f"Latency:    p50 = {percentile(latencies, 50):.0f} ms   "
              f"p95 = {percentile(latencies, 95):.0f} ms   (successful calls only)")
    else:
        print("Latency:    no successful calls")
    if outcome.get("internal_errors"):
        print(f"WARNING:    {outcome['internal_errors']} internal runner errors")
    if errors:
        print(f"To retry the failed cases:  audit run --resume {run_id}")
    print("=" * 50)
    return len(errors)


# ---------------------------------------------------------------- audit run
def cmd_run(args: argparse.Namespace) -> int:
    cfg = load_config(args.target)
    for w in cfg.warnings():
        print(f"warning: {w}")

    testset_path = args.testset or cfg.paths.testset
    cases = load_testset(testset_path)
    if args.limit is not None:
        cases = cases[: args.limit]
    if not cases:
        print("No test cases to run.")
        return 1

    store = Store(cfg.paths.database)

    run_id = args.resume or f"run_{datetime.now():%Y%m%d_%H%M%S}_{uuid.uuid4().hex[:6]}"
    if args.resume and store.get_run(run_id) is None:
        print(f"error: no run called '{run_id}' in {cfg.paths.database}")
        return 1

    # results point at cases, so the cases must be in the database first.
    # add_cases replaces rows with the same id, so edited cases are refreshed.
    store.add_cases(cases)

    target = build_target(cfg)
    snapshot = {**target.config_snapshot(), "models": cfg.models.model_dump()}
    concurrency = args.concurrency or cfg.budget.max_concurrency

    print(f"Run {run_id}: {len(cases)} cases, concurrency {concurrency}, "
          f"target '{cfg.target.name}' ({cfg.target.type})")

    async def go() -> dict:
        ticker = asyncio.create_task(_show_progress(store, run_id, len(cases)))
        try:
            return await run_audit(
                store, target, cases, run_id=run_id, config_snapshot=snapshot,
                concurrency=concurrency, timeout_s=cfg.budget.timeout_s)
        finally:
            ticker.cancel()

    try:
        outcome = asyncio.run(go())
    except KeyboardInterrupt:
        print(f"\nStopped. Finished cases are saved. Continue with:  audit run --resume {run_id}")
        return 130

    n_errors = print_summary(store, run_id, {c.id for c in cases}, outcome)
    return 0 if n_errors == 0 else 2


# ---------------------------------------------------------------- audit check
def cmd_check(args: argparse.Namespace) -> int:
    """Run the deterministic checks on every saved answer of a run.

    Each answer gets one row in the `checks` table. Empty and error answers are
    reported as their own status and are never turned into a score of 0.
    """
    cfg = load_config(args.target)
    store = Store(cfg.paths.database)

    if store.get_run(args.run) is None:
        print(f"error: no run called '{args.run}' in {cfg.paths.database}")
        return 1

    cases = {c.id: c for c in store.get_cases()}
    results = store.get_results(args.run)
    if not results:
        print("No results to check.")
        return 1

    by_status: Counter = Counter()
    by_expected: Counter = Counter()
    for r in results:
        chk = run_checks(r.answer, bot_error=r.error)
        store.add_check(args.run, r.case_id, chk.status, chk.has_citation, chk.detail)
        by_status[chk.status] += 1
        case = cases.get(r.case_id)
        expected = case.expected_behavior.value if case else "?"
        by_expected[(expected, chk.status)] += 1

    print(f"Run {args.run}: {len(results)} answers checked")
    for status, n in sorted(by_status.items()):
        print(f"  {status:10} {n}")
    print("By expected behavior:")
    for (expected, status), n in sorted(by_expected.items()):
        print(f"  {expected:8} {status:10} {n}")
    return 0


# ---------------------------------------------------------------- parser
def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="audit", description="LLM audit toolkit")
    sub = parser.add_subparsers(dest="command", required=True)

    run = sub.add_parser("run", help="ask the target every test question and save the answers")
    run.add_argument("--target", default="configs/audit.yaml",
                     help="path to the audit YAML (default: configs/audit.yaml)")
    run.add_argument("--testset", default=None,
                     help="path to the .jsonl test cases (default: paths.testset in the YAML)")
    run.add_argument("--concurrency", type=int, default=None,
                     help="cases asked at the same time (default: budget.max_concurrency)")
    run.add_argument("--resume", metavar="RUN_ID", default=None,
                     help="continue this run: skip finished cases, retry failed ones")
    run.add_argument("--limit", type=int, default=None, metavar="N",
                     help="only run the first N cases")
    run.set_defaults(func=cmd_run)

    chk = sub.add_parser("check", help="run deterministic checks on a run's saved answers")
    chk.add_argument("--run", required=True, metavar="RUN_ID", help="the run to check")
    chk.add_argument("--target", default="configs/audit.yaml",
                     help="path to the audit YAML (default: configs/audit.yaml)")
    chk.set_defaults(func=cmd_check)
    return parser


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    if getattr(args, "concurrency", None) is not None and args.concurrency < 1:
        print("error: --concurrency must be at least 1")
        return 1
    if getattr(args, "limit", None) is not None and args.limit < 1:
        print("error: --limit must be at least 1")
        return 1
    try:
        return args.func(args)
    except (FileNotFoundError, ValueError, ValidationError) as e:
        print(f"error: {e}")
        return 1


if __name__ == "__main__":
    sys.exit(main())
